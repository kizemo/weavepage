// slug 生成 + 去重 — 纯函数,不依赖 DOM/Tiptap
export function slugify(text: string, index: number): string {
  const base = text
    .toLowerCase()
    .replace(/[^\p{L}\p{N}\s-]/gu, "")
    .replace(/\s+/g, "-")
    .replace(/^-+|-+$/g, "")
    .slice(0, 60);
  return base || `n${index}`;
}

export function uniqueSlug(base: string, taken: Set<string>): string {
  let s = base;
  let n = 2;
  while (taken.has(s)) {
    s = `${base}-${n++}`;
  }
  taken.add(s);
  return s;
}
```

export interface TocEntry {
  level: 2 | 3;
  text: string;
  id: string;
}

export function scanHeadings(root: HTMLElement): TocEntry[] {
  const out: TocEntry[] = [];
  const taken = new Set<string>();
  const nodes = root.querySelectorAll("h2, h3");
  nodes.forEach((el, idx) => {
    const tag = el.tagName.toLowerCase();
    if (tag !== "h2" && tag !== "h3") return;
    const level = (tag === "h2" ? 2 : 3) as 2 | 3;
    const text = (el.textContent ?? "").trim();
    if (!text) return;
    let id = el.id || "";
    if (!id) {
      const base = slugify(text, idx + 1);
      id = `wp-toc-${uniqueSlug(base, taken)}`;
    } else {
      taken.add(id);
    }
    out.push({ level, text, id });
  });
  return out;
}
```

const HTML_ESC: Record<string, string> = {
  "&": "&amp;",
  "<": "&lt;",
  ">": "&gt;",
  '"': "&quot;",
};
const escHtml = (s: string): string =>
  s.replace(/[&<>"]/g, (c) => HTML_ESC[c] ?? c);

export function buildDocTocHtml(entries: TocEntry[]): string {
  if (entries.length === 0) {
    return `<nav class="doc-toc doc-toc-empty" contenteditable="false">`
      + `<div class="doc-toc-title">目录 ▾</div>`
      + `<div class="doc-toc-list-wrap"><div class="doc-toc-empty-msg">未找到任何标题</div></div>`
      + `</nav>`;
  }
  // 嵌套:h2 直接子 <li>,h3 嵌在最近 h2 的子 <ul>
  const openLis: string[] = [];
  const closeLis: string[] = [];
  let inner = "";
  let liOpen = false;
  let inSub = false;
  for (let i = 0; i < entries.length; i++) {
    const e = entries[i]!;
    if (e.level === 2) {
      if (liOpen) {
        // 关闭上一个 h2
        inner += inSub ? "</ul></li>" : "</li>";
        inSub = false;
      }
      inner += `<li><a class="doc-toc-h2" href="#${escHtml(e.id)}">${escHtml(e.text)}</a>`;
      liOpen = true;
      // 检查下一个是否 h3
      const next = entries[i + 1];
      if (next && next.level === 3) {
        inner += "<ul>";
        inSub = true;
      } else {
        inner += "</li>";
        liOpen = false;
      }
    } else {
      // h3 — 必须有 liOpen
      if (!liOpen) continue;
      inner += `<li><a class="doc-toc-h3" href="#${escHtml(e.id)}">${escHtml(e.text)}</a></li>`;
      // 若下一个不是 h3,关闭 sub
      const next = entries[i + 1];
      if (!next || next.level !== 3) {
        if (inSub) {
          inner += "</ul></li>";
          inSub = false;
          liOpen = false;
        }
      }
    }
  }
  // 处理尾部未关闭
  if (liOpen) {
    inner += inSub ? "</ul></li>" : "</li>";
  }
  void openLis; void closeLis; // 占位防 lint
  return (
    `<nav class="doc-toc" contenteditable="false">` +
    `<div class="doc-toc-title">目录 ▾</div>` +
    `<div class="doc-toc-list-wrap"><ul class="doc-toc-list">${inner}</ul></div>` +
    `</nav>`
  );
}
```

export function assignHeadingIds(root: HTMLElement, entries: TocEntry[]): void {
  const nodes = Array.from(root.querySelectorAll("h2, h3"));
  entries.forEach((e, i) => {
    const el = nodes[i];
    if (!el) return;
    if (el.id !== e.id) {
      if (e.id) el.setAttribute("id", e.id);
      else el.removeAttribute("id");
    }
  });
}

export function clearHeadingIds(root: HTMLElement): void {
  root.querySelectorAll("h2, h3").forEach((el) => {
    if (el.id.startsWith("wp-toc-")) el.removeAttribute("id");
  });
}
```

```ts
import type { Editor } from "@tiptap/core";

// 通过 ProseMirror 视图 DOM 找到 nav.doc-toc 节点;返回其 PM pos(若没有返回 null)
const findDocTocPos = (editor: Editor): number | null => {
  const dom = editor.view.dom as HTMLElement;
  const navEl = dom.querySelector("nav.doc-toc");
  if (!navEl) return null;
  const pmView = editor.view;
  try {
    const pos = pmView.posAtDOM(navEl, 0);
    return pos;
  } catch {
    return null;
  }
};

// 删除 nav.doc-toc 节点(含其内容)
const removeDocToc = (editor: Editor): boolean => {
  const pos = findDocTocPos(editor);
  if (pos == null) return false;
  const $pos = editor.state.doc.resolve(pos);
  // 找到包含 nav.doc-toc 的最高节点
  let depth = $pos.depth;
  let node = $pos.node(depth);
  while (depth > 0) {
    const el = editor.view.nodeDOM($pos.before(depth));
    if (el === null || !(el instanceof HTMLElement)) break;
    if (!el.classList.contains("doc-toc")) {
      depth--;
      node = $pos.node(depth);
      continue;
    }
    break;
  }
  const from = $pos.before(depth);
  const to = from + node.nodeSize;
  editor.chain().focus().deleteRange({ from, to }).run();
  return true;
};

const insertDocToc = (editor: Editor, html: string): void => {
  editor.chain().focus().insertContentAt(0, html).run();
};

export function scanTocEntries(editor: Editor): TocEntry[] {
  const dom = editor.view.dom as HTMLElement;
  return scanHeadings(dom);
}

export function regenDocToc(editor: Editor): TocEntry[] {
  const entries = scanTocEntries(editor);
  // 写锚点
  const dom = editor.view.dom as HTMLElement;
  assignHeadingIds(dom, entries);
  // 替换 nav
  const html = buildDocTocHtml(entries);
  removeDocToc(editor);
  if (entries.length > 0) {
    insertDocToc(editor, html);
  } else {
    // 空 nav 也不插入(只保留按钮提示),但仍然要 clear existing nav(removeDocToc 已做)
  }
  return entries;
}

export function enableDocToc(editor: Editor): TocEntry[] {
  const entries = scanTocEntries(editor);
  const dom = editor.view.dom as HTMLElement;
  assignHeadingIds(dom, entries);
  if (entries.length === 0) {
    // 即便没标题,插入空提示 nav 让用户能看到「目录」存在
    insertDocToc(editor, buildDocTocHtml(entries));
  } else {
    removeDocToc(editor);
    insertDocToc(editor, buildDocTocHtml(entries));
  }
  return entries;
}

export function disableDocToc(editor: Editor): void {
  removeDocToc(editor);
  const dom = editor.view.dom as HTMLElement;
  clearHeadingIds(dom);
}
```

(function(){
  if (window.__wpTocInited) return; window.__wpTocInited = true;

  // 1. 创建悬浮按钮 + 面板(wrapper 共享 hover)
  var wrap = document.createElement('div');

  wrap.style.cssText = 'position:fixed;top:80px;right:24px;z-index:900;';

  btn.title = '目录';
  wrap.appendChild(btn);

  var panel = document.createElement('div');
  panel.className = 'wp-toc-floating-panel';
  wrap.appendChild(panel);

  // 2. 扫描 h2 / h3 渲染嵌套 ul
  var render = function(){
    var heads = document.querySelectorAll('h2, h3');
    var html = '<div class="doc-toc-title">目录 ▾</div><div class="doc-toc-list-wrap"><ul class="doc-toc-list">';
    var openLi = false, inSub = false;
    for (var i = 0; i < heads.length; i++){
      var h = heads[i];
      var tag = h.tagName.toLowerCase();
      if (tag !== 'h2' && tag !== 'h3') continue;
      var txt = (h.textContent || '').trim();
      if (!txt) continue;
      var id = h.id || ('wp-toc-' + (i+1));
      if (tag === 'h2'){
        if (openLi) html += inSub ? '</ul></li>' : '</li>';
        html += '<li class="wp-toc-floating-item"><a class="doc-toc-h2" href="#' + id + '">' + txt + '</a>';
        openLi = true;
        var next = heads[i+1];
        if (next && next.tagName.toLowerCase() === 'h3'){
          html += '<ul>';
          inSub = true;
        } else {
          html += '</li>';
          openLi = false;
        }
      } else {
        if (!openLi) continue;
        html += '<li class="wp-toc-floating-item"><a class="doc-toc-h3" href="#' + id + '">' + txt + '</a></li>';
        var next2 = heads[i+1];
        if (!next2 || next2.tagName.toLowerCase() !== 'h3'){
          if (inSub){ html += '</ul></li>'; inSub = false; openLi = false; }
        }
      }
    }
    if (openLi) html += inSub ? '</ul></li>' : '</li>';
    html += '</ul></div>';
    panel.innerHTML = html;
  };
  render();

  // 3. hover 展开 / 250ms 延迟折叠
  var closeTimer = null;
  var openPanel = function(){
    if (closeTimer){ clearTimeout(closeTimer); closeTimer = null; }
    btn.classList.add('is-open');
    panel.style.display = 'block';
  };
  var scheduleClose = function(){
    if (closeTimer) clearTimeout(closeTimer);
    closeTimer = setTimeout(function(){
      btn.classList.remove('is-open');
      panel.style.display = 'none';

      if (t.tagName === 'A' && t.getAttribute('href') && t.getAttribute('href').charAt(0) === '#'){

        var tgt = document.getElementById(id);
        if (tgt){ tgt.scrollIntoView({ behavior: 'smooth', block: 'start' }); }
        e.preventDefault();
        return;
      }
      t = t.parentNode;
    }
  });

  // 5. IntersectionObserver 高亮当前标题
  if ('IntersectionObserver' in window){
    var obs = new IntersectionObserver(function(items){
      items.forEach(function(it){
        var item = it.target;

            }
          });
        }
      });
    }, { rootMargin: '0px 0px -70% 0px', threshold: 0 });
    document.querySelectorAll('h2, h3').forEach(function(h){ if (h.id) obs.observe(h); });
  }
})();
</script>`;

import { useEffect, useRef } from "react";
import { createPortal } from "react-dom";
import type { Editor } from "@tiptap/core";
import type { TocEntry } from "../utils/tocGen";

interface TocFloatingProps {
  editor: Editor;
  entries: TocEntry[];
  enabled: boolean;
}

export function TocFloating({ editor, entries, enabled }: TocFloatingProps) {
  const wrapRef = useRef<HTMLDivElement>(null);
  const closeTimerRef = useRef<number | null>(null);

  useEffect(() => {
    const wrap = wrapRef.current;
    if (!wrap) return;
    const open = () => {
      if (closeTimerRef.current) {
        clearTimeout(closeTimerRef.current);
        closeTimerRef.current = null;
      }
      wrap.classList.add("is-open");
    };
    const scheduleClose = () => {
      if (closeTimerRef.current) clearTimeout(closeTimerRef.current);
      closeTimerRef.current = window.setTimeout(() => {
        wrap.classList.remove("is-open");
      }, 250);
    };
    wrap.addEventListener("mouseenter", open);
    wrap.addEventListener("mouseleave", scheduleClose);
    return () => {
      wrap.removeEventListener("mouseenter", open);
      wrap.removeEventListener("mouseleave", scheduleClose);
    };
  }, []);

  const handleClick = (id: string) => (e: React.MouseEvent) => {
    e.preventDefault();
    const dom = editor.view.dom as HTMLElement;
    const tgt = dom.querySelector(`#${CSS.escape(id)}`);
    if (tgt) {
      tgt.scrollIntoView({ behavior: "smooth", block: "start" });
      setTimeout(() => editor.commands.focus(), 200);
    }
  };

  if (!enabled) return null;

  // 渲染嵌套 ul:h3 嵌在最近 h2 的子 ul
  const renderEntries = (): React.ReactNode[] => {
    const out: React.ReactNode[] = [];
    let openLi = false;
    let inSub = false;
    entries.forEach((e, i) => {
      if (e.level === 2) {
        if (openLi) {
          out.push(inSub ? (
            <ul key={"sub-close-" + i}>{/* 闭合占位 — 实际由 React fragment 处理 */}</ul>
          ) : null);
        }
        // 简化:用 flat array + key,React 自动闭合 ul
      }
    });
    return out;
  };
  // 上面是占位逻辑,实际用更干净的递归渲染:
  const renderTree = (level: 2 | 3, startIdx: number): { nodes: React.ReactNode[]; nextIdx: number } => {
    const nodes: React.ReactNode[] = [];
    let i = startIdx;
    while (i < entries.length) {
      const e = entries[i]!;
      if (e.level === level) {
        const children = level === 2
          ? (() => {
              const next = entries[i + 1];
              if (next && next.level === 3) {
                const sub = renderTree(3, i + 1);
                i = sub.nextIdx;
                return sub.nodes;
              }
              return null;
            })()
          : null;
        nodes.push(
          <li key={e.id} className="wp-toc-floating-item">
            <a className={`doc-toc-h${e.level}`} href={`#${e.id}`} onClick={handleClick(e.id)}>
              {e.text}
            </a>
            {children && <ul className="wp-toc-floating-submenu">{children}</ul>}
          </li>
        );
        i++;
      } else if (level === 2 && e.level === 3) {
        // 已在 h2 子节点里处理;此处跳过
        i++;
      } else {
        break;
      }
    }
    return { nodes, nextIdx: i };
  };
  const { nodes } = renderTree(2, 0);
  void renderEntries;

  return createPortal(
    <div id="wp-toc-floating-wrap" className="wp-toc-floating-wrap" ref={wrapRef}>
      <button className="wp-toc-floating" title="目录">📑</button>
      <div className="wp-toc-floating-panel">
        <div className="doc-toc-title">目录 ▾</div>
        <div className="doc-toc-list-wrap">
          <ul className="doc-toc-list">{nodes}</ul>
        </div>
      </div>
    </div>,
    document.body
  );
}

### Task 12: App.tsx 集成 tocEnabled + 调度 regen(debounce + flush)

- 调度 regen:
  - `editor.on('transaction')` → docChanged 且 transaction.getMeta('tocRegenPending') → scheduleRegen(400ms debounce)
  - `editor.on('selectionUpdate')` → 若 selection 离开标题节点且 pending → flush

打开 `src/App.tsx`,在 `editor = useEditor({...})` 之后追加:

```tsx
// ---- TOC 状态 ----
const tocEnabled = activeDoc.tocEnabled === true;
const [tocEntries, setTocEntries] = useState<TocEntry[]>([]);
const regenPendingRef = useRef(false);
const regenTimerRef = useRef<number | null>(null);

const refreshTocEntries = useCallback(() => {
  if (!editor || !tocEnabled) {
    setTocEntries([]);
    return;
  }
  setTocEntries(scanTocEntries(editor));
}, [editor, tocEnabled]);

const flushRegen = useCallback(() => {
  if (!editor || !tocEnabled) return;
  if (regenTimerRef.current) {
    clearTimeout(regenTimerRef.current);
    regenTimerRef.current = null;
  }
  regenDocToc(editor);
  refreshTocEntries();
  regenPendingRef.current = false;
}, [editor, tocEnabled, refreshTocEntries]);

const scheduleRegen = useCallback(() => {
  if (!editor || !tocEnabled) return;
  regenPendingRef.current = true;
  if (regenTimerRef.current) clearTimeout(regenTimerRef.current);
  regenTimerRef.current = window.setTimeout(() => {
    regenTimerRef.current = null;
    flushRegen();
  }, 400);
}, [editor, tocEnabled, flushRegen]);

// 监听 editor 事件
useEffect(() => {
  if (!editor || !tocEnabled) return;
  const onTransaction = ({ transaction }: { transaction: any }) => {
    if (!transaction.docChanged) return;
    // 只在标题节点变化时 scheduleRegen
    // 简单实现:任何 docChanged 都 schedule(避免漏)
    scheduleRegen();
  };
  const onSelectionUpdate = () => {
    if (!regenPendingRef.current) return;
    const { selection } = editor.state;
    const $head = selection.$head;
    let inHeading = false;
    for (let d = $head.depth; d > 0; d--) {
      if ($head.node(d).type.name === "heading") { inHeading = true; break; }
    }
    if (!inHeading) flushRegen();
  };
  const onBlur = () => flushRegen();
  editor.on("transaction", onTransaction);
  editor.on("selectionUpdate", onSelectionUpdate);
  editor.view.dom.addEventListener("blur", onBlur);
  return () => {
    editor.off("transaction", onTransaction);
    editor.off("selectionUpdate", onSelectionUpdate);
    editor.view.dom.removeEventListener("blur", onBlur);
  };
}, [editor, tocEnabled, scheduleRegen, flushRegen]);

// 切文档前 flush(switchDoc / closeDoc / newDoc 之前)
const flushBeforeSwitch = useCallback(() => {
  if (regenTimerRef.current) {
    clearTimeout(regenTimerRef.current);
    regenTimerRef.current = null;
  }
  if (regenPendingRef.current) {
    regenDocToc(editor!);
    regenPendingRef.current = false;
  }
}, [editor]);

// 切换/关闭/新建前 flush(在已有 switchDoc / closeDoc / newDoc 开头调)
useEffect(() => {
  // 包装:snapshotActive 之前 flush
  const origSnapshot = snapshotActive;
  // ... 此处略 — 实际通过改 snapshotActive 来注入 flush
  // 简化:在 switchDoc / closeDoc / newDoc 内部调 flushBeforeSwitch
  void origSnapshot;
}, []);

// 启用 / 禁用 TOC
const onToggleToc = useCallback(() => {
  if (!editor) return;
  if (tocEnabled) {
    disableDocToc(editor);
    updateActiveDoc({ tocEnabled: false });
    setTocEntries([]);
  } else {
    enableDocToc(editor);
    updateActiveDoc({ tocEnabled: true });
    setTocEntries(scanTocEntries(editor));
  }
}, [editor, tocEnabled, updateActiveDoc]);

// activeId 切换后刷新 tocEntries
useEffect(() => {
  refreshTocEntries();
}, [activeId, refreshTocEntries]);
```
